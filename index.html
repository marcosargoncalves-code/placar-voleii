    <script>

        let data = { pointsA: 0, pointsB: 0, setsA: 0, setsB: 0 };


        function changeValue(field, amount) {

            data[field] += amount;

            if (data[field] < 0) data[field] = 0;

            document.getElementById(field + '-display').innerText = data[field];

        }


        function updateNames() {

            let nameA = document.getElementById('nameA-input').value;

            let nameB = document.getElementById('nameB-input').value;

            // Limita a exibição a 3 letras maiúsculas para manter o estilo condensado da VNL

            document.getElementById('teamA-display').innerText = nameA.substring(0, 4).toUpperCase();

            document.getElementById('teamB-display').innerText = nameB.substring(0, 4).toUpperCase();

        }


        function updateLogos() {

            let urlA = document.getElementById('urlA-input').value;

            let urlB = document.getElementById('urlB-input').value;

            document.getElementById('logoA-display').src = urlA;

            document.getElementById('logoB-display').src = urlB;

        }


        function updateCenterLogo() {

            document.getElementById('logo-display').innerText = document.getElementById('logo-input').value;

        }


        function resetScoreboard() {

            data = { pointsA: 0, pointsB: 0, setsA: 0, setsB: 0 };

            document.getElementById('pointsA-display').innerText = 0;

            document.getElementById('pointsB-display').innerText = 0;

            document.getElementById('setsA-display').innerText = 0;

            document.getElementById('setsB-display').innerText = 0;

        }
